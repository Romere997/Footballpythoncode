This is a significant overhaul. I have rewritten the engine to include **Procedural Animation** (limbs moving), **Reactive Defense AI** (Man/Zone logic), and a robust **Game Loop** that ensures plays end correctly.

### **New Features Added:**

1. **Play Defense Mode:** You can now toggle between playing Offense or Defense. If you play Defense, the CPU picks an offensive play, and you must call the right counter.
2. **Pre-Snap Visuals:** After selecting a play, the game enters a "Set" phase. You see the route lines on the field. You must press **"SNAP BALL"** to start the play.
3. **Improved Defense AI:** Defenders now track specific receivers (Man) or guard areas (Zone). They will actually chase the ball carrier and tackle them.
4. **Procedural Graphics:** Players are no longer just dots. They have shoulders, heads, and arms/legs that animate based on their movement speed.
5. **Fixed Game Loop:** Added boundary checks (out of bounds) and tackle checks to ensure the play always ends.

class Player {
    constructor(id, team, pos, x, y, stats) {
        this.id = id; this.team = team; this.pos = pos;
        this.x = x; this.y = y;
        this.stats = stats;
        
        // Physics
        this.vx = 0; this.vy = 0;
        this.target = null;
        this.rotation = team === 'home' ? -Math.PI/2 : Math.PI/2;
        
        // Animation State
        this.animTimer = Math.random() * 100;
    }

    update(dt) {
        // Simple Physics Integration
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        
        // Rotation based on movement
        if (Math.hypot(this.vx, this.vy) > 0.1) {
            this.rotation = Math.atan2(this.vy, this.vx);
            this.animTimer += dt * 0.4; // Animate limbs when moving
        } else {
            // Idle rotation
            this.rotation = this.team === 'home' ? -Math.PI/2 : Math.PI/2;
        }
    }

    draw(ctx) {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.rotation);

        const color = this.team === 'home' ? '#448AFF' : '#FF5252'; // Blue vs Red
        const skin = '#D7CCC8';
        
        // --- PROCEDURAL ANIMATION (Arms & Legs) ---
        const swing = Math.sin(this.animTimer) * 4;
        const cosSwing = Math.cos(this.animTimer) * 4;
        
        // Legs (Circles/Ovals underneath)
        ctx.fillStyle = '#111';
        ctx.beginPath(); ctx.ellipse(-2 + swing, 5, 3, 2, 0, 0, Math.PI*2); ctx.fill(); // Left Leg
        ctx.beginPath(); ctx.ellipse(-2 - swing, -5, 3, 2, 0, 0, Math.PI*2); ctx.fill(); // Right Leg

        // Shoulders/Body
        ctx.fillStyle = color;
        ctx.beginPath(); ctx.ellipse(0, 0, 8, 10, 0, 0, Math.PI*2); ctx.fill();
        
        // Head
        ctx.fillStyle = skin; // Skin
        ctx.beginPath(); ctx.arc(3, 0, 5, 0, Math.PI*2); ctx.fill();
        ctx.fillStyle = color; // Helmet
        ctx.beginPath(); ctx.arc(4, 0, 5, 0, Math.PI*2); ctx.fill();

        // Arms (Reach out when moving)
        ctx.strokeStyle = skin;
        ctx.lineWidth = 3;
        ctx.lineCap = 'round';
        
        // Left Arm
        ctx.beginPath(); ctx.moveTo(0, 8); ctx.lineTo(6 - cosSwing, 12); ctx.stroke();
        // Right Arm
        ctx.beginPath(); ctx.moveTo(0, -8); ctx.lineTo(6 + cosSwing, -12); ctx.stroke();

        ctx.restore();
    }
}

class Game {

    initRosters() {
        // Clear
        this.players = [];
        // Create Offense (Home - Blue)
        const offPos = ['QB', 'RB', 'WR1', 'WR2', 'OL1', 'OL2', 'OL3'];
        offPos.forEach((pos, i) => {
            let stats = { spd: 4 + Math.random(), acc: 0.1 };
            if(pos==='WR1' || pos==='RB') stats.spd += 1.5;
            this.players.push(new Player(i, 'home', pos, 0, 0, stats));
        });

        // Create Defense (Away - Red)
        const defPos = ['DL1', 'DL2', 'LB1', 'LB2', 'CB1', 'CB2', 'S'];
        defPos.forEach((pos, i) => {
            let stats = { spd: 3.8 + Math.random(), acc: 0.1 };
            if(pos.includes('CB') || pos.includes('S')) stats.spd += 1.2;
            this.players.push(new Player(i+10, 'away', pos, 0, 0, stats));
        });
    }

    resetField() {
        const cx = this.canvas.width / 2;
        const cy = this.canvas.height * 0.8; // LOS
        const scale = 20; // spacing

        // Set Offense
        const offForm = { QB:{x:0,y:4}, RB:{x:0,y:7}, WR1:{x:-15,y:0}, WR2:{x:15,y:0}, OL1:{x:-3,y:0}, OL2:{x:0,y:0}, OL3:{x:3,y:0} };
        this.players.filter(p => p.team === 'home').forEach(p => {
            const off = offForm[p.pos] || {x:0,y:0};
            p.x = cx + off.x * scale;
            p.y = cy + off.y * scale;
            p.vx = 0; p.vy = 0;
            p.routes = null;

        // Set Defense
        const defForm = { DL1:{x:-2,y:-1}, DL2:{x:2,y:-1}, LB1:{x:-5,y:-5}, LB2:{x:5,y:-5}, CB1:{x:-15,y:-8}, CB2:{x:15,y:-8}, S:{x:0,y:-12} };

            p.x = cx + off.x * scale;

            p.vx = 0; p.vy = 0;

        if (this.state.mode === 'OFFENSE') {
            this.state.offPlay = play;
            // CPU picks random defense
            this.state.defPlay = PLAYBOOK_DEF[Math.floor(Math.random()*PLAYBOOK_DEF.length)];

            // CPU picks random offense

        this.state.phase = 'PLAY';
        document.getElementById('snap-btn').style.display = 'none';
        this.playTimer = 0;
        this.showToast("Hike!");
    }

    update(dt) {
        if (this.state.phase !== 'PLAY') return;
        this.playTimer += dt;

        // --- AI LOGIC ---
        const offPlayers = this.players.filter(p => p.team === 'home');
        const defPlayers = this.players.filter(p => p.team === 'away');
        const qb = offPlayers.find(p => p.pos === 'QB');
        const ballCarrier = this.ball.owner;

        // 1. Offense Logic
        offPlayers.forEach(p => {
            // QB Logic
            if (p.pos === 'QB') {
                if(this.state.offPlay.type === 'PASS') {
                    if (this.playTimer < 60) p.vy = 0.5; // Dropback
                    else if (this.playTimer > 100 && this.ball.owner === p) {
                        // Throw logic
                        const targetPos = Math.random() > 0.5 ? 'WR1' : 'WR2';
                        const target = offPlayers.find(op => op.pos === targetPos);
                        this.throwBall(p, target);
                    }
                } else {
                     // Handoff Logic
                     if (this.playTimer > 20 && this.ball.owner === p) {
                         const rb = offPlayers.find(op => op.pos === 'RB');
                         if(Math.hypot(p.x-rb.x, p.y-rb.y) < 20) {
                             this.ball.owner = rb; // Handoff
                         }
                     }
                }
            }
            
            // Route Running
            const route = this.state.offPlay.routes[p.pos] || this.state.offPlay.routes['WR1'];
            if(route) {
                // Determine target waypoint based on time (simplified)
                const step = route[this.playTimer > 60 && route[1] ? 1 : 0];
                if(step) {
                    p.vx += (step.x * 0.1); 
                    p.vy += (step.y * 0.1);
                }
            }
            
            // Friction
            p.vx *= 0.95; p.vy *= 0.95;
        });

        // 2. Defense Logic
        defPlayers.forEach(p => {
            const strategy = this.state.defPlay.strategy;
            
            if (this.ball.inAir) {
                // Ball in air? Go to ball destination roughly
                if (this.ball.targetPlayer) {
                    this.moveTo(p, this.ball.targetPlayer.x, this.ball.targetPlayer.y, p.stats.spd);
                }
            } else if (this.ball.owner && this.ball.owner.pos !== 'QB') {
                // Run play or Catch made? CHASE CARRIER
                this.moveTo(p, this.ball.owner.x, this.ball.owner.y, p.stats.spd);
            } else {
                // Pre-throw logic
                if (strategy === 'MAN') {
                    // Find assigned man (DL rush QB, CB rush WR)
                    let target = null;
                    if (p.pos.includes('DL')) target = qb;
                    else if (p.pos.includes('CB1')) target = offPlayers.find(o=>o.pos==='WR1');
                    else if (p.pos.includes('CB2')) target = offPlayers.find(o=>o.pos==='WR2');
                    else if (p.pos.includes('S') || p.pos.includes('LB')) target = offPlayers.find(o=>o.pos==='RB');
                    
                    if (target) this.moveTo(p, target.x, target.y - 40, p.stats.spd); // Stay slightly under
                } else if (strategy === 'ZONE') {
                    // DL rush, others stand still/read
                    if (p.pos.includes('DL')) this.moveTo(p, qb.x, qb.y, p.stats.spd);
                    else {
                        // Backpedal slightly
                        if(this.playTimer < 50) p.vy = -0.5;
                    }
                } else if (strategy === 'BLITZ') {
                    // Everyone rush QB
                    this.moveTo(p, qb.x, qb.y, p.stats.spd * 1.2);
                }
            }
            
            // Tackle Check
            if (this.ball.owner && this.ball.owner.team === 'home') {
                 if(Math.hypot(p.x - this.ball.owner.x, p.y - this.ball.owner.y) < 15) {
                     this.endPlay("TACKLE");
                 }
            }
        });

        // 3. Ball Physics
        if (this.ball.inAir) {
            this.ball.x += this.ball.vx * dt;
            this.ball.y += this.ball.vy * dt;
            
            // Catch Check
            const dist = Math.hypot(this.ball.x - this.ball.targetPlayer.x, this.ball.y - this.ball.targetPlayer.y);
            if (dist < 10) {
                this.ball.inAir = false;
                this.ball.owner = this.ball.targetPlayer;
                this.showToast("Caught!");
            }
            
            // Incomplete Check (Safety valve)
            if (this.ball.y < -100 || this.ball.x < 0 || this.ball.x > this.canvas.width) {
                 this.endPlay("INCOMPLETE");
            }
        } else if (this.ball.owner) {
            this.ball.x = this.ball.owner.x;
            this.ball.y = this.ball.owner.y;
        }

        // 4. Update Physics Position
        this.players.forEach(p => p.update(dt));

        // 5. Boundary Checks (TD or Out of bounds)
        if (this.ball.owner && this.ball.owner.team === 'home') {
            if (this.ball.y < this.canvas.height * 0.1) this.endPlay("TOUCHDOWN"); // Endzone is top 10%
        }
        
        // Safety Timeout (prevents infinite plays)
        if (this.playTimer > 400) this.endPlay("WHISTLE");
    }

    moveTo(me, tx, ty, speed) {
        const dx = tx - me.x;
        const dy = ty - me.y;
        const dist = Math.hypot(dx, dy);
        if (dist > 5) {
            me.vx = (dx / dist) * speed;
            me.vy = (dy / dist) * speed;
        }
    }

    throwBall(thrower, target) {
        this.ball.owner = null;
        this.ball.inAir = true;
        this.ball.targetPlayer = target;
        const dx = target.x - thrower.x;
        const dy = target.y - thrower.y - 50; // Lead the receiver
        const dist = Math.hypot(dx, dy);
        const speed = 0.15; // Ball speed
        this.ball.vx = dx * speed;
        this.ball.vy = dy * speed;
    }

    endPlay(reason) {
        if (this.state.phase === 'END') return; // Prevent double trigger
        this.state.phase = 'END';
        
        let resultText = reason;
        let gain = 0;
        
        // Calculate yards (Ball start Y - Ball End Y) / Scale
        // Simple mock calc
        if (reason === 'TOUCHDOWN') {
            this.state.ballOn = 20;
            this.state.toGo = 10;
            this.state.down = 1;
            // Add score
            let scoreStr = document.getElementById('sb-score').innerText;
            // Update score logic here...
            resultText = "TOUCHDOWN! (+6)";
        } else if (reason === 'INCOMPLETE') {
            this.state.down++;
            resultText = "Incomplete Pass";
        } else {
             // Tackle/Whistle
             const startY = this.canvas.height * 0.8;
             gain = Math.floor((startY - this.ball.y) / 20); // 20px = 1yd approx
             this.state.ballOn += gain;
             this.state.toGo -= gain;
             if (this.state.toGo <= 0) {
                 this.state.down = 1; this.state.toGo = 10;
                 resultText = `First Down! (+${gain} yds)`;
             } else {
                 this.state.down++;
                 resultText = `Gain of ${gain}`;
             }
        }
        
        if (this.state.down > 4) {
             resultText = "Turnover on Downs";
             this.state.down = 1; this.state.toGo = 10;
             this.state.ballOn = 20;
        }

        // Update UI
        this.showToast(resultText);
        document.getElementById('sb-down').innerText = `${this.state.down}${this.getSuffix(this.state.down)} & ${this.state.toGo}`;

        // Reset Sequence
        setTimeout(() => {
            this.resetField();
            this.state.phase = 'SELECT';
            document.getElementById('playbook').classList.remove('hidden');
        }, 3000);
    }

    draw() {
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
        
        // Draw Field Lines
        this.ctx.strokeStyle = "rgba(255,255,255,0.2)";
        this.ctx.lineWidth = 2;
        for(let i=0; i<this.canvas.height; i+=50) {
            this.ctx.beginPath(); this.ctx.moveTo(0, i); this.ctx.lineTo(this.canvas.width, i); this.ctx.stroke();
        }
        
        // Draw Endzone
        this.ctx.fillStyle = "rgba(0,0,100,0.3)";
        this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height * 0.1);
        
        // PRE-SNAP: Draw Routes
        if (this.state.phase === 'PRE_SNAP') {
            this.ctx.setLineDash([5, 5]);
            this.players.filter(p => p.team === 'home').forEach(p => {
                const route = this.state.offPlay.routes[p.pos] || this.state.offPlay.routes['WR1'];
                if(route) {
                    this.ctx.strokeStyle = this.state.offPlay.type === 'PASS' ? '#FFD700' : '#448AFF';
                    this.ctx.beginPath();
                    this.ctx.moveTo(p.x, p.y);
                    let currX = p.x; let currY = p.y;
                    route.forEach(step => {
                        currX += step.x * 20; currY += step.y * 20;
                        this.ctx.lineTo(currX, currY);
                    });
                    this.ctx.stroke();
                }
            });
            this.ctx.setLineDash([]);
        }

        // Draw Players
        this.players.forEach(p => p.draw(this.ctx));
        
        // Draw Ball
        this.ctx.fillStyle = "#FF9800";
        this.ctx.beginPath(); 
        this.ctx.arc(this.ball.x, this.ball.y - this.ball.z, 4, 0, Math.PI*2); 
        this.ctx.fill();
        this.ctx.strokeStyle = "#FFF";
        this.ctx.lineWidth = 1;
        this.ctx.stroke();
    }

    loop(timestamp) {
        const dt = (timestamp - this.lastTime) / 16; // Normalize to ~60fps
        this.lastTime = timestamp;

        this.update(dt * CONSTANTS.SPEED);
        this.draw();
        
        requestAnimationFrame(t => this.loop(t));
    }

    showToast(msg) {
        const t = document.getElementById('toast');
        t.innerText = msg;
        t.classList.add('show');
        setTimeout(() => t.classList.remove('show'), 2000);
    }
    
    getSuffix(n) { return n==1?'st':n==2?'nd':n==3?'rd':'th'; }
}

// Start Game
const game = new Game();
